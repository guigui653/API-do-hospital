import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;

@Configuration
@EnableWebSecurity
public class SecurityConfigurations {

    @Autowired
    private SecurityFilter securityFilter;

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        return http
                .cors(cors -> cors.configurationSource(request -> {
                    var corsConfiguration = new org.springframework.web.cors.CorsConfiguration();
                    corsConfiguration.setAllowedOriginPatterns(java.util.List.of("*"));
                    corsConfiguration.setAllowedMethods(java.util.List.of("GET", "POST", "PUT", "DELETE", "OPTIONS", "PATCH", "HEAD"));
                    corsConfiguration.setAllowedHeaders(java.util.List.of("*"));
                    corsConfiguration.setAllowCredentials(true);
                    return corsConfiguration;
                }))
                .csrf(csrf -> csrf.disable())
                .sessionManagement(sm -> sm.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
                .authorizeHttpRequests(req -> {
                    req.requestMatchers(HttpMethod.OPTIONS, "/**").permitAll();
                    req.requestMatchers("/triagem.html", "/fila.html", "/static/**", "/css/**", "/js/**", "/images/**").permitAll();

package com.hospital.api.infra.security;

import com.hospital.api.repository.UsuarioRepository;
import com.hospital.api.repository.PacienteRepository;
import com.hospital.api.entity.Usuario;
import com.hospital.api.entity.Paciente;
import com.hospital.api.enums.TipoUsuario;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;

@Component
public class SecurityFilter extends OncePerRequestFilter {

    @Autowired
    private TokenService tokenService;

    @Autowired
    private UsuarioRepository usuarioRepository;

    @Autowired
    private PacienteRepository pacienteRepository;

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain filterChain) throws ServletException, IOException {
        var tokenJWT = recuperarToken(request);

        if (tokenJWT != null) {
            try {
                var subject = tokenService.getSubject(tokenJWT);

                org.springframework.security.core.userdetails.UserDetails userDetails = usuarioRepository.findByCpf(subject)
                        .filter(u -> u.getTipo() != TipoUsuario.PACIENTE)
                        .map(u -> (org.springframework.security.core.userdetails.UserDetails) u)
                        .orElse(null);

                if (userDetails == null) {
                    userDetails = pacienteRepository.findByCpf(subject).orElse(null);
                }
                if (userDetails == null && subject.length() == 11) {
                    String subjectFormatado = subject.substring(0, 3) + "." +
                                              subject.substring(3, 6) + "." +
                                              subject.substring(6, 9) + "-" +
                                              subject.substring(9, 11);
                    userDetails = pacienteRepository.findByCpf(subjectFormatado).orElse(null);
                }

                if (userDetails != null) {
                    var authentication = new UsernamePasswordAuthenticationToken(userDetails, null, userDetails.getAuthorities());
                    SecurityContextHolder.getContext().setAuthentication(authentication);
                }
            } catch (RuntimeException e) {
                // Token inválido — deixa passar sem autenticação
            }
        }

        filterChain.doFilter(request, response);
    }

    private String recuperarToken(HttpServletRequest request) {
        var authorizationHeader = request.getHeader("Authorization");
        if (authorizationHeader != null && authorizationHeader.startsWith("Bearer ")) {
            var token = authorizationHeader.replace("Bearer ", "").trim();
            return token.isBlank() ? null : token;
        }
        return null;
    }
}

package com.hospital.api.infra.security;

import com.auth0.jwt.JWT;
import com.auth0.jwt.algorithms.Algorithm;
import com.auth0.jwt.exceptions.JWTCreationException;
import com.auth0.jwt.exceptions.JWTVerificationException;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import java.time.Instant;
import java.time.LocalDateTime;
import java.time.ZoneOffset;

@Service
public class TokenService {

    @Value("${api.security.token.secret}")
    private String secret;

    public String gerarToken(String subject) {
        try {
            var algoritmo = Algorithm.HMAC256(secret);
            return JWT.create()
                    .withIssuer("API AppCuidar+")
                    .withSubject(subject)
                    .withExpiresAt(dataExpiracao())
                    .sign(algoritmo);
        } catch (JWTCreationException exception){
            throw new RuntimeException("Erro ao gerar token jwt", exception);
        }
    }

    public String getSubject(String tokenJWT) {
        try {
            var algoritmo = Algorithm.HMAC256(secret);
            return JWT.require(algoritmo)
                    .withIssuer("API AppCuidar+")
                    .build()
                    .verify(tokenJWT)
                    .getSubject();
        } catch (JWTVerificationException exception) {
            throw new RuntimeException("Token JWT inválido ou expirado!");
        }
    }

    private Instant dataExpiracao() {
        return LocalDateTime.now().plusDays(30).toInstant(ZoneOffset.of("-03:00"));
    }
}

package com.hospital.api.controller;

import com.hospital.api.dto.request.DadosAutenticacaoDTO;
import com.hospital.api.dto.response.TokenDTO;
import com.hospital.api.entity.Paciente;
import com.hospital.api.entity.Usuario;
import com.hospital.api.infra.security.TokenService;
import com.hospital.api.repository.PacienteRepository;
import jakarta.validation.Valid;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.authentication.AuthenticationCredentialsNotFoundException;
import org.springframework.security.authentication.AuthenticationManager;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.AuthenticationException;
import org.springframework.web.bind.annotation.*;

@RestController
@CrossOrigin(originPatterns = "*")
public class AutenticacaoController {

    @Autowired
    private AuthenticationManager manager;

    @Autowired
    private TokenService tokenService;

    @Autowired
    private PacienteRepository pacienteRepository;

    @PostMapping({"/pacientes/login", "/api/auth/login"})
    public ResponseEntity<TokenDTO> efetuarLogin(@RequestBody @Valid DadosAutenticacaoDTO dados) {
        try {
            var cpf = dados.cpf().replaceAll("\\D", "");
            if (cpf.isBlank()) {
                throw new AuthenticationCredentialsNotFoundException("CPF vazio");
            }

            var authenticationToken = new UsernamePasswordAuthenticationToken(cpf, dados.senha());
            var authentication = manager.authenticate(authenticationToken);

            var principal = authentication.getPrincipal();
            String tokenJWT;
            Long idPaciente = null;
            String nome;
            String leito = null;

            if (principal instanceof Usuario usuario) {
                tokenJWT = tokenService.gerarToken(usuario.getCpf());
                nome = usuario.getNome();

                Paciente paciente = pacienteRepository.findByCpf(usuario.getCpf()).orElse(null);
                if (paciente != null) {
                    idPaciente = paciente.getIdPaciente();
                    leito = paciente.getLeito();
                }
            } else if (principal instanceof Paciente paciente) {
                tokenJWT = tokenService.gerarToken(paciente.getCpf());
                idPaciente = paciente.getIdPaciente();
                nome = paciente.getNome();
                leito = paciente.getLeito();
            } else {
                throw new AuthenticationCredentialsNotFoundException("Principal de autenticação inválido.");
            }

            return ResponseEntity.ok(new TokenDTO(tokenJWT, idPaciente, nome, leito));

        } catch (AuthenticationException e) {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED).build();
        }
    }
}

package com.hospital.api.controller;

import com.hospital.api.dto.request.DadosTriagemDTO;

import com.hospital.api.enums.NivelUrgencia;
import com.hospital.api.repository.TriagemRepository;
import com.hospital.api.service.TriagemService;
import jakarta.validation.Valid;
import org.springframework.http.ResponseEntity;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.bind.annotation.*;

import java.util.Comparator;
import java.util.List;

@RestController
@RequestMapping("/api/triagem")
@CrossOrigin(originPatterns = "*")
public class TriagemController {

    private final TriagemRepository triagemRepository;
    private final TriagemService triagemService;

    public TriagemController(TriagemRepository triagemRepository, TriagemService triagemService) {
        this.triagemRepository = triagemRepository;
        this.triagemService = triagemService;
    }

    @PostMapping
    @Transactional
    public ResponseEntity<TriagemResponseDTO> receberTriagem(@RequestBody @Valid DadosTriagemDTO dados) {
        TriagemResponseDTO resposta = triagemService.registrarTriagem(dados);
        return ResponseEntity.ok(resposta);
    }

    @GetMapping
    public ResponseEntity<List<DadosListagemTriagemDTO>> listar() {
        var lista = triagemRepository.findAll()
                .stream()
                .map(DadosListagemTriagemDTO::new)
                .sorted(Comparator.comparingInt(this::pesoUrgencia))
                .toList();
        return ResponseEntity.ok(lista);
    }

    @DeleteMapping("/{id}")
    @Transactional
    public ResponseEntity<Void> atender(@PathVariable Long id) {
        if (!triagemRepository.existsById(id)) {
            return ResponseEntity.notFound().build();
        }
        triagemRepository.deleteById(id);
        return ResponseEntity.noContent().build();
    }

    private int pesoUrgencia(DadosListagemTriagemDTO dto) {
        if (dto.urgencia() == null) return 4;
        return switch (dto.urgencia()) {
            case VERMELHO -> 1;
            case AMARELO  -> 2;
            case VERDE    -> 3;
        };
    }
}

@Service
public class PacienteService {

    private final PacienteRepository pacienteRepository;
    private final PasswordEncoder passwordEncoder;

    public PacienteService(PacienteRepository pacienteRepository, PasswordEncoder passwordEncoder) {
        this.pacienteRepository = pacienteRepository;
        this.passwordEncoder = passwordEncoder;
    }

    @Transactional
    public PacienteResponseDTO cadastrar(PacienteRequestDTO dto) {
        if (dto.cpf() == null || dto.cpf().isBlank()) {
            throw new IllegalArgumentException("Erro: O CPF é obrigatório para cadastrar o paciente!");
        }
        var cpfLimpo = dto.cpf().replaceAll("\\D", "");
        if (cpfLimpo.length() < 11) {
            throw new IllegalArgumentException("Erro: CPF inválido! Digite todos os 11 números.");
        }

        // Senha inicial: primeiros 3 dígitos do CPF
        String senhaInicial = cpfLimpo.substring(0, 3);

        Paciente paciente = new Paciente();
        paciente.setNome(dto.nome());
        paciente.setCpf(cpfLimpo);
        paciente.setSenha(passwordEncoder.encode(senhaInicial));
        paciente.setTelefoneContato(dto.telefoneContato());
        paciente.setNumeroProntuario(dto.numeroProntuario());
        paciente.setDataNascimento(dto.dataNascimento());
        paciente.setSexo(dto.sexo());
        paciente.setNomeResponsavel(dto.nomeResponsavel());
        paciente.setTelefoneResponsavel(dto.telefoneResponsavel());
        paciente.setAtivo(true);

        pacienteRepository.save(paciente);
        return new PacienteResponseDTO(paciente);
    }

    @Transactional(readOnly = true)
    public List<PacienteResponseDTO> listarTodos() {
        return pacienteRepository.findAll().stream().map(PacienteResponseDTO::new).toList();
    }

    @Transactional(readOnly = true)
    public PacienteResponseDTO buscarPorId(Long id) {
        Paciente paciente = pacienteRepository.findById(id)
                .orElseThrow(() -> new RuntimeException("Paciente não encontrado"));
        return new PacienteResponseDTO(paciente);
    }

    @Transactional
    public PacienteResponseDTO atualizar(Long id, PacienteAtualizacaoDTO dto) {
        Paciente paciente = pacienteRepository.findById(id)
                .orElseThrow(() -> new RuntimeException("Paciente não encontrado para atualização"));
        if (dto.telefoneContato() != null) paciente.setTelefoneContato(dto.telefoneContato());
        if (dto.nomeResponsavel() != null) paciente.setNomeResponsavel(dto.nomeResponsavel());
        if (dto.telefoneResponsavel() != null) paciente.setTelefoneResponsavel(dto.telefoneResponsavel());
        pacienteRepository.save(paciente);
        return new PacienteResponseDTO(paciente);
    }

package com.hospital.api.service;

import com.hospital.api.dto.request.EsqueciSenhaRequestDTO;
import com.hospital.api.dto.request.ResetarSenhaRequestDTO;
import com.hospital.api.entity.RecuperacaoSenha;
import com.hospital.api.entity.Usuario;
import com.hospital.api.entity.Paciente;
import com.hospital.api.repository.RecuperacaoSenhaRepository;
import com.hospital.api.repository.UsuarioRepository;
import com.hospital.api.repository.PacienteRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.security.SecureRandom;
import java.time.LocalDateTime;
import java.util.Map;

@Slf4j
@Service
@RequiredArgsConstructor
public class RecuperacaoSenhaService {

    private static final int EXPIRACAO_MINUTOS = 15;

    private final UsuarioRepository usuarioRepository;
    private final PacienteRepository pacienteRepository;
    private final RecuperacaoSenhaRepository recuperacaoSenhaRepository;
    private final PasswordEncoder passwordEncoder;

    @Transactional
    public Map<String, String> solicitarRecuperacao(EsqueciSenhaRequestDTO dto) {
        String cpfLimpo = dto.cpf().replaceAll("\\D", "");

        var usuarioOpt = usuarioRepository.findByCpf(cpfLimpo);
        if (usuarioOpt.isPresent()) {
            Usuario usuario = usuarioOpt.get();
            recuperacaoSenhaRepository.findByUsuarioAndUtilizadoFalse(usuario)
                    .ifPresent(tokenAntigo -> {
                        tokenAntigo.setUtilizado(true);
                        recuperacaoSenhaRepository.save(tokenAntigo);
                    });

            String codigo = gerarCodigoDe6Digitos();
            RecuperacaoSenha recuperacao = new RecuperacaoSenha();
            recuperacao.setUsuario(usuario);
            recuperacao.setCodigoConfirmacao(codigo);
            recuperacao.setDataExpiracao(LocalDateTime.now().plusMinutes(EXPIRACAO_MINUTOS));
            recuperacao.setUtilizado(false);
            recuperacaoSenhaRepository.save(recuperacao);

            log.info("[SIMULACAO EMAIL] Codigo para {}: {}", usuario.getEmail(), codigo);
            return Map.of("emailMascarado", mascararEmail(usuario.getEmail()));
        }

        var pacienteOpt = pacienteRepository.findByCpf(cpfLimpo);
        if (pacienteOpt.isPresent()) {
            Paciente paciente = pacienteOpt.get();
            recuperacaoSenhaRepository.findByPacienteAndUtilizadoFalse(paciente)
                    .ifPresent(tokenAntigo -> {
                        tokenAntigo.setUtilizado(true);
                        recuperacaoSenhaRepository.save(tokenAntigo);
                    });

            String codigo = gerarCodigoDe6Digitos();
            RecuperacaoSenha recuperacao = new RecuperacaoSenha();
            recuperacao.setPaciente(paciente);
            recuperacao.setCodigoConfirmacao(codigo);
            recuperacao.setDataExpiracao(LocalDateTime.now().plusMinutes(EXPIRACAO_MINUTOS));
            recuperacao.setUtilizado(false);
            recuperacaoSenhaRepository.save(recuperacao);

            String emailPaciente = cpfLimpo + "@paciente.hospital.com";
            log.info("[SIMULACAO EMAIL] Codigo para {}: {}", emailPaciente, codigo);
            return Map.of("emailMascarado", mascararEmail(emailPaciente));
        }

        throw new RuntimeException("Nenhum usuário ou paciente encontrado com este CPF.");
    }

    @Transactional
    public void resetarSenha(ResetarSenhaRequestDTO dto) {
        RecuperacaoSenha recuperacao = recuperacaoSenhaRepository
                .findByCodigoConfirmacao(dto.codigo())
                .orElseThrow(() -> new RuntimeException("Código de verificação inválido."));

        if (recuperacao.getUtilizado()) {
            throw new RuntimeException("Este código já foi utilizado. Solicite um novo.");
        }
        if (LocalDateTime.now().isAfter(recuperacao.getDataExpiracao())) {
            throw new RuntimeException("Código expirado. Solicite um novo código de verificação.");
        }

        if (recuperacao.getUsuario() != null) {
            Usuario usuario = recuperacao.getUsuario();
            usuario.setSenha(passwordEncoder.encode(dto.novaSenha()));
            usuarioRepository.save(usuario);
        } else if (recuperacao.getPaciente() != null) {
            Paciente paciente = recuperacao.getPaciente();
            paciente.setSenha(passwordEncoder.encode(dto.novaSenha()));
            pacienteRepository.save(paciente);
        }

        recuperacao.setUtilizado(true);
        recuperacaoSenhaRepository.save(recuperacao);
    }

    private String gerarCodigoDe6Digitos() {
        SecureRandom random = new SecureRandom();
        int numero = 100000 + random.nextInt(900000);
        return String.valueOf(numero);
    }

    private String mascararEmail(String email) {
        if (email == null || !email.contains("@")) return "***@***.com";
        String[] partes = email.split("@");
        String usuario = partes[0];
        String dominio = partes[1];
        if (usuario.length() <= 1) return email;
        return usuario.charAt(0) + "****@" + dominio;
    }
}

    private final PacienteRepository pacienteRepository;
    private final LeitoRepository leitoRepository;

                          PacienteRepository pacienteRepository,
                          LeitoRepository leitoRepository) {
        this.chamadoRepository = chamadoRepository;
        this.pacienteRepository = pacienteRepository;
        this.leitoRepository = leitoRepository;
    }

    @Transactional
    public ChamadoResponseDTO abrirChamado(ChamadoRequestDTO dto) {
        Paciente paciente = resolverPaciente(dto.idPaciente());
        Chamado chamado = new Chamado();
        chamado.setPaciente(paciente);

        Leito leito = null;
        if (dto.idLeito() != null) {
            leito = leitoRef(dto.idLeito());
        } else if (paciente.getLeito() != null && !paciente.getLeito().isBlank()) {
            leito = leitoRepository.findByNumero(paciente.getLeito()).orElse(null);
        }
        chamado.setLeito(leito);
        chamado.setServico(dto.idServico() != null ? servicoRef(dto.idServico()) : null);
        chamado.setObservacao(dto.observacao() != null ? dto.observacao() : "Chamado aberto via app.");
        chamado.setStatus(StatusChamado.ABERTO);
        chamado.setPrioridade("MEDIA");
        chamado.setDataAbertura(LocalDateTime.now());
        chamado.setTipo(dto.tipo() != null ? dto.tipo() : TipoChamado.ENFERMAGEM);

        chamadoRepository.save(chamado);
        return new ChamadoResponseDTO(chamado);
    }

    @Transactional
    public ChamadoResponseDTO confirmarAtendimento(Long idChamado) {
        Chamado chamado = chamadoRepository.findById(idChamado)
                .orElseThrow(() -> new RuntimeException("Chamado #" + idChamado + " não encontrado."));
        if (chamado.getStatus() != StatusChamado.ABERTO) {
            throw new RuntimeException("Chamado #" + idChamado + " já está " + chamado.getStatus().name());
        }
        chamado.setStatus(StatusChamado.EM_ATENDIMENTO);
        chamado.setDataInicioAtend(LocalDateTime.now());
        chamadoRepository.save(chamado);
        return new ChamadoResponseDTO(chamado);
    }

    @Transactional
    public ChamadoResponseDTO finalizarAtendimento(Long idChamado, FinalizarChamadoRequestDTO dto) {
        Chamado chamado = chamadoRepository.findById(idChamado)
                .orElseThrow(() -> new RuntimeException("Chamado #" + idChamado + " não encontrado."));
        if (chamado.getStatus() != StatusChamado.EM_ATENDIMENTO) {
            throw new RuntimeException("Apenas chamados EM_ATENDIMENTO podem ser finalizados.");
        }
        chamado.setStatus(StatusChamado.CONCLUIDO);
        chamado.setDescricaoResolucao(dto.descricaoResolucao());
        chamado.setDataConclusao(LocalDateTime.now());
        chamadoRepository.save(chamado);
